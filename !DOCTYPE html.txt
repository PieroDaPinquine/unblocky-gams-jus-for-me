    <script>
        // CHANGE THESE TWO LINES
        const USER = "YOUR_USERNAME"; 
        const REPO = "YOUR_REPO_NAME";
        const BASE_URL = `https://cdn.jsdelivr.net/gh/${USER}/${REPO}@main/`;

        async function loadGame() {
            const status = document.getElementById('status');
            const bar = document.getElementById('bar');
            let parts = [];

            try {
                for (let i = 1; i <= 13; i++) {
                    status.innerText = `Grabbing Ingredients... (${i}/13)`;
                    const res = await fetch(`${BASE_URL}game.unx.part${i}`);
                    if (!res.ok) throw new Error("Delivery Failed");
                    parts.push(await res.arrayBuffer());
                    bar.style.width = `${(i / 13) * 100}%`;
                }

                status.innerText = "Baking Pizza...";
                const fullGame = new Blob(parts, { type: 'application/octet-stream' });
                const gameUrl = URL.createObjectURL(fullGame);

                // Override Fetch so the engine finds the merged file
                const oldFetch = window.fetch;
                window.fetch = (url, config) => {
                    if (typeof url === 'string' && url.includes("game.unx")) {
                        return Promise.resolve(new Response(fullGame));
                    }
                    return oldFetch(url, config);
                };

                // Load the actual engine scripts
                const scripts = ["index.js", "runner.js"];
                for (let s of scripts) {
                    await new Promise(r => {
                        let sc = document.createElement("script");
                        sc.src = BASE_URL + s;
                        sc.onload = r;
                        document.body.appendChild(sc);
                    });
                }

                setTimeout(() => document.getElementById('loader').remove(), 2000);
            } catch (e) {
                status.innerText = "Error: Kitchen is closed (403/404)";
                console.error(e);
            }
        }
        loadGame();
    </script>
